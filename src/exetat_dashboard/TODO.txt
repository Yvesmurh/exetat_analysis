2.top Performing province being changed with (top Performing student )
  Top Performing student for each option for the current years
  comparaison of percentage of current student with the one for last years 

WITH ranked_2023 AS (
    SELECT
        o.id AS option_id,
        o.option_name,
        s.id AS student_id,
        s.student_name,
        er.percentage,
        ROW_NUMBER() OVER (PARTITION BY o.id ORDER BY er.percentage DESC) AS rn
    FROM exam_results er
    JOIN exetat_entry ee ON er.school_session_id = ee.id
    JOIN options o ON ee.option_id = o.id
    JOIN students s ON er.student_id = s.id
    WHERE ee.years = 2023
),
ranked_2022 AS (
    SELECT
        o.id AS option_id,
        MAX(er.percentage) AS top_percentage_2022
    FROM exam_results er
    JOIN exetat_entry ee ON er.school_session_id = ee.id
    JOIN options o ON ee.option_id = o.id
    WHERE ee.years = 2022
    GROUP BY o.id
)
SELECT
    r.option_name,
    r.student_name AS top_student_2023,
    r.percentage AS top_percentage_2023,
    p.top_percentage_2022,
    ROUND(r.percentage - p.top_percentage_2022, 2) AS pts_change
FROM ranked_2023 r
LEFT JOIN ranked_2022 p ON r.option_id = p.option_id
WHERE r.rn = 1
ORDER BY r.option_name;

4.province performance map
these are datas that will be display on the Map ,For every province:

  Difference (by rate of success) compared to the last year

WITH prov_2023 AS (
    SELECT
        p.id AS province_id,
        p.name AS province_name,
        COUNT(DISTINCT sc.id) AS num_schools,
        SUM(ee.participants) AS num_participants,
        SUM(ee.reussites) AS num_successes,
        ROUND(SUM(ee.reussites) / NULLIF(SUM(ee.participants), 0) * 100, 2) AS success_rate_2023
    FROM pr_after_reform26 p
    JOIN ed_province ep ON ep.province_after_id = p.id
    JOIN schools sc ON sc.province_id = ep.id
    JOIN exetat_entry ee ON ee.school_id = sc.id AND ee.years = 2020
    GROUP BY p.id, p.name
),
prov_2022 AS (
    SELECT
        p.id AS province_id,
        ROUND(SUM(ee.reussites) / NULLIF(SUM(ee.participants), 0) * 100, 2) AS success_rate_2022
    FROM pr_after_reform26 p
    JOIN ed_province ep ON ep.province_after_id = p.id
    JOIN schools sc ON sc.province_id = ep.id
    JOIN exetat_entry ee ON ee.school_id = sc.id AND ee.years = 2019
    GROUP BY p.id
)
SELECT
    c.province_name,
    c.success_rate_2023,
    c.num_schools,
    c.num_participants,
    c.num_successes,
    p.success_rate_2022,
    ROUND(c.success_rate_2023 - p.success_rate_2022, 2) AS rate_change_pts
FROM prov_2023 c
LEFT JOIN prov_2022 p ON c.province_id = p.province_id
ORDER BY c.province_name;

SELECT
    p.id as province_id,
    p.name AS province_name,
    COUNT(DISTINCT sc.id) AS num_schools,
    SUM(ee.participants) AS num_participants,
    SUM(ee.reussites) AS num_successes,
    ROUND(SUM(ee.reussites) / NULLIF(SUM(ee.participants), 0) * 100, 2) AS success_rate
FROM pr_after_reform26 p
JOIN ed_province ep ON ep.province_after_id = p.id
JOIN schools sc ON sc.province_id = ep.id
JOIN exetat_entry ee ON ee.school_id = sc.id AND ee.years = 2020
GROUP BY p.id, p.name
ORDER BY success_rate DESC;
